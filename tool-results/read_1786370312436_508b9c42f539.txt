    36→function formatTime(seconds: number): string {
    37→  if (!isFinite(seconds) || isNaN(seconds)) return "0:00";
    38→  const m = Math.floor(seconds / 60);
    39→  const s = Math.floor(seconds % 60);
    40→  return `${m}:${s.toString().padStart(2, "0")}`;
    41→}
    42→
    43→function formatTime12(time24: string): string {
    44→  const [h, m] = time24.split(":").map(Number);
    45→  const ampm = h >= 12 ? "PM" : "AM";
    46→  const h12 = h % 12 || 12;
    47→  return `${h12}:${m.toString().padStart(2, "0")} ${ampm}`;
    48→}

   510→  const calcQibla = (lat: number, lng: number, _locName?: string) => {
   511→    const kaabaLat = (21.4225 * Math.PI) / 180;
   512→    const kaabaLng = (39.8262 * Math.PI) / 180;
   513→    const latR = (lat * Math.PI) / 180;
   514→    const lngR = (lng * Math.PI) / 180;
   515→    const y = Math.sin(kaabaLng - lngR);
   516→    const x = Math.cos(latR) * Math.tan(kaabaLat) - Math.sin(latR) * Math.cos(kaabaLng - lngR);
   517→    let qibla = (Math.atan2(y, x) * 180) / Math.PI;
   518→    qibla = ((qibla % 360) + 360) % 360;
   519→    setQiblaAngle(qibla);
