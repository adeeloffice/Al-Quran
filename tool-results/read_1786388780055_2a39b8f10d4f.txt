    41→    41→function formatTime(seconds: number): string {
    42→    42→  if (!isFinite(seconds) || isNaN(seconds)) return "0:00";
    43→    43→  const m = Math.floor(seconds / 60);
    44→    44→  const s = Math.floor(seconds % 60);
    45→    45→  return `${m}:${s.toString().padStart(2, "0")}`;
    46→    46→}
    47→    47→
    48→    48→function formatTime12(time24: string): string {
    49→    49→  const [h, m] = time24.split(":").map(Number);
    50→    50→  const ampm = h >= 12 ? "PM" : "AM";
    51→    51→  const h12 = h % 12 || 12;
    52→    52→  return `${h12}:${m.toString().padStart(2, "0")} ${ampm}`;
    53→    53→}

    59→    59→function toArabicNumeral(num: number): string {
    60→    60→  const arabicDigits = ["٠", "١", "٢", "٣", "٤", "٥", "٦", "٧", "٨", "٩"];
    61→    61→  return num.toString().split("").map(d => arabicDigits[parseInt(d)]).join("");
    62→    62→}
