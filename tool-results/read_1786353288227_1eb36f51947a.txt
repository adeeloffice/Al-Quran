    34→function formatTime(seconds: number): string {
    35→  if (!isFinite(seconds) || isNaN(seconds)) return "0:00";
    36→  const m = Math.floor(seconds / 60);
    37→  const s = Math.floor(seconds % 60);
    38→  return `${m}:${s.toString().padStart(2, "0")}`;
    39→}
    40→
    41→function formatTime12(time24: string): string {
    42→  const [h, m] = time24.split(":").map(Number);
    43→  const ampm = h >= 12 ? "PM" : "AM";
    44→  const h12 = h % 12 || 12;
    45→  return `${h12}:${m.toString().padStart(2, "0")} ${ampm}`;
    46→}

    48→function proxyUrl(originalUrl: string): string {
    49→  return `/api/audio?url=${encodeURIComponent(originalUrl)}`;
