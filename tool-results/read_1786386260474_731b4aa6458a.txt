    39→    39→function formatTime(seconds: number): string {
    40→    40→  if (!isFinite(seconds) || isNaN(seconds)) return "0:00";
    41→    41→  const m = Math.floor(seconds / 60);
    42→    42→  const s = Math.floor(seconds % 60);
    43→    43→  return `${m}:${s.toString().padStart(2, "0")}`;
    44→    44→}
    45→    45→
    46→    46→function formatTime12(time24: string): string {
    47→    47→  const [h, m] = time24.split(":").map(Number);
    48→    48→  const ampm = h >= 12 ? "PM" : "AM";
    49→    49→  const h12 = h % 12 || 12;
    50→    50→  return `${h12}:${m.toString().padStart(2, "0")} ${ampm}`;
    51→    51→}
