    36→function formatTime(seconds: number): string {
    37→  if (!isFinite(seconds) || isNaN(seconds)) return "0:00";
    38→  const m = Math.floor(seconds / 60);
    39→  const s = Math.floor(seconds % 60);
    40→  return `${m}:${s.toString().padStart(2, "0")}`;
    41→}
    42→
    43→function formatTime12(time24: string): string {
    44→  const [h, m] = time24.split(":").map(Number);
    45→  const ampm = h >= 12 ? "PM" : "AM";
    46→  const h12 = h % 12 || 12;
    47→  return `${h12}:${m.toString().padStart(2, "0")} ${ampm}`;
    48→}

    50→function proxyUrl(originalUrl: string): string {

   473→    let qibla = (Math.atan2(y, x) * 180) / Math.PI;
