    41→function formatTime(seconds: number): string {
    42→  if (!isFinite(seconds) || isNaN(seconds)) return "0:00";
    43→  const m = Math.floor(seconds / 60);
    44→  const s = Math.floor(seconds % 60);
    45→  return `${m}:${s.toString().padStart(2, "0")}`;
    46→}
    47→
    48→function formatTime12(time24: string): string {
    49→  const [h, m] = time24.split(":").map(Number);
    50→  const ampm = h >= 12 ? "PM" : "AM";
    51→  const h12 = h % 12 || 12;
    52→  return `${h12}:${m.toString().padStart(2, "0")} ${ampm}`;
    53→}

    55→function proxyUrl(originalUrl: string): string {

    59→function toArabicNumeral(num: number): string {
    60→  const arabicDigits = ["٠", "١", "٢", "٣", "٤", "٥", "٦", "٧", "٨", "٩"];
    61→  return num.toString().split("").map(d => arabicDigits[parseInt(d)]).join("");
    62→}

   526→  const calcQibla = (lat: number, lng: number, _locName?: string) => {
   527→    const kaabaLat = (21.4225 * Math.PI) / 180;
   528→    const kaabaLng = (39.8262 * Math.PI) / 180;
   529→    const latR = (lat * Math.PI) / 180;
   530→    const lngR = (lng * Math.PI) / 180;
   531→    const y = Math.sin(kaabaLng - lngR);
   532→    const x = Math.cos(latR) * Math.tan(kaabaLat) - Math.sin(latR) * Math.cos(kaabaLng - lngR);
   533→    let qibla = (Math.atan2(y, x) * 180) / Math.PI;
   534→    qibla = ((qibla % 360) + 360) % 360;
   535→    setQiblaAngle(qibla);
